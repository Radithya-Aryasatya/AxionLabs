217|         /// </summary>
218|         /// <param name="jsonInstance">The instance in simplified presentation.</param>

220|         public static Instance FromJsonInstance(JsonInstance jsonInstance)
221|         {
222|             // Create instance with given parameters
223|             Instance instance = new Instance
224|             {
225|                 Name = jsonInstance.Name,
226|                 Data = jsonInstance.Data,
227|             };
228|             instance.Containers.AddRange(jsonInstance.Containers.Select(c => new Container()
229|             {
230|                 ID = c.ID,
231|                 Mesh = new MeshCube()
232|                 {
233|                     Length = c.Length,
234|                     Width = c.Width,
235|                     Height = c.Height
236|                 },
237|                 MaxWeight = c.MaxWeight,
238|                 Data = c.Data
239|             }));
240|             instance.Pieces.AddRange(jsonInstance.Pieces.Select(p =>
241|             {
242|                 var convp = new VariablePiece() { ID = p.ID, Weight = p.Weight };

244|                     convp.SetFlags(p.Flags.Select(f => (f.FlagId, f.FlagValue)));
245| 
246|                 if (p.ForbiddenOrientations != null && p.AllowedOrientations == null)
247|                     convp.ForbiddenOrientations = new(p.ForbiddenOrientations);

251|                     if (p.ForbiddenOrientations != null)
252|                         convp.ForbiddenOrientations.UnionWith(p.ForbiddenOrientations);

259|                 return convp;
260|             }));
