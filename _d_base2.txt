   1| """
   2| solver_sardine.baseline
   3| =======================
   4| Pure-Python port of the SardineCan **Extreme Point Insertion** heuristic.
   5| 
   6| The engine consumes the *same* manifest shape that ``app.py`` builds and
   7| returns a :class:`~solver_common.schemas.Solution` whose ``packed`` list
   8| is converted by ``app.py`` into ``PackedItem`` objects — so the existing
   9| 3-D viewer, metrics, and fleet registration work unchanged.
  10| 
  11| Algorithm (mirrors
  12| ``SC.Core/Heuristics/PrimalHeuristic/ExtremePointInsertion.cs``):
  13| 
  14| 1. **Extreme Points (EPs).** The container starts with the origin corner
  15|    as a candidate placement point. After every successful placement the
  16|    8 corners of the newly-placed box are added as EPs.
  17| 
  18| 2. **Best-fit merit.** For each (piece, EP, rotation) combination the
  19|    *minimum residual slack* across all three axes is computed. The
  20|    placement with the smallest residual (tightest fit) wins.
  21| 
  22| 3. **Guard rails.** Before acceptance a candidate must pass:
  23|    bounds, collision, stability (>= 0.75 support ratio), weight,
  24|    and the orientation rule: ``updown=False`` items may only assume
  25|    ``RT_WHD`` (0) or ``RT_DHW`` (3) — spin on floor, never tip.
  26| 
  27| 4. **Soft-LIFO (optional).** When ``prioritize_sequence`` is True items
  28|    with a higher unloading sequence are placed first (deepest). An item
  29|    may never be placed where an already-fitted higher-sequence box sits
  30|    between it and the door in the same lane and shelf.
  31| 
  32| 5. **EP pruning.** After each placement, EPs interior to a placed box or
  33|    dominated by another EP are removed.
  34| """

  71| def _rot_dim(w, h, d, rt):
  72|     """Return (width, height, depth) for a given rotation type."""
  73|     if rt == RotationType.RT_WHD:   # 0
  74|         return w, h, d
  75|     if rt == RotationType.RT_HWD:   # 1
  76|         return h, w, d
  77|     if rt == RotationType.RT_HDW:   # 2
  78|         return h, d, w
  79|     if rt == RotationType.RT_DHW:   # 3
  80|         return d, h, w
  81|     if rt == RotationType.RT_DWH:   # 4
  82|         return d, w, h
  83|     if rt == RotationType.RT_WDH:   # 5
  84|         return w, d, h
  85|     return w, h, d

  88| def _aabb_overlap(a, b):
  89|     """3-D AABB overlap. a/b = (x,y,z,w,h,d)."""
  90|     ax0, ay0, az0 = a[0], a[1], a[2]
  91|     ax1, ay1, az1 = ax0 + a[3], ay0 + a[4], az0 + a[5]
  92|     bx0, by0, bz0 = b[0], b[1], b[2]
  93|     bx1, by1, bz1 = bx0 + b[3], by0 + b[4], bz0 + b[5]
  94|     return (ax0 < bx1 - EPS and bx0 < ax1 - EPS and
  95|             ay0 < by1 - EPS and by0 < ay1 - EPS and
  96|             az0 < bz1 - EPS and bz0 < az1 - EPS)
  97| 
  98| 
  99| def _xz_overlap(a, b):
 100|     """2-D X-Z overlap area (footprint projection)."""
 101|     ax0, az0 = a[0], a[2]
 102|     ax1, az1 = ax0 + a[3], az0 + a[5]
 103|     bx0, bz0 = b[0], b[2]
 104|     bx1, bz1 = bx0 + b[3], bz0 + b[5]
 105|     ox = min(ax1, bx1) - max(ax0, bx0)
 106|     oz = min(az1, bz1) - max(az0, bz0)
 107|     return ox * oz if ox > 0 and oz > 0 else 0.0

 114| def _is_supported(x, y, z, w, h, d, placed, support_ratio):
 115|     """Stability check — same rule as ``Bin._isStableAt``.
 116| 
 117|     * y ≤ 0 → on floor, always stable.
 118|     * Otherwise ≥ ``support_ratio`` of the footprint must be covered by the
 119|       top faces of already-placed items directly beneath, *and* the centre
 120|       of mass must lie on a supporter.  Below 0.25 we allow a 4-vertex fallback.
 121|     """
 122|     if y <= EPS:
 123|         return True
 124|     footprint = w * d
 125|     if footprint <= 0:
 126|         return True
 127|     support_area = 0.0
 128|     centre_ok = False
 129|     cx = x + w / 2.0
 130|     cz = z + d / 2.0
 131|     for (px, py, pz, pw, ph, pd) in placed:
 132|         if abs(y - (py + ph)) > 1e-3:
 133|             continue
 134|         ox = min(x + w, px + pw) - max(x, px)
 135|         oz = min(z + d, pz + pd) - max(z, pz)
 136|         if ox > EPS and oz > EPS:
 137|             support_area += ox * oz
 138|             if px <= cx <= px + pw and pz <= cz <= pz + pd:
 139|                 centre_ok = True
 140|     ratio = support_area / footprint
 141|     if ratio >= support_ratio:
 142|         return centre_ok
 143|     if ratio < 0.25:
 144|         return False
 145|     # 4-vertex fallback
 146|     four = [(x, z), (x + w, z), (x, z + d), (x + w, z + d)]
 147|     covered = [False] * 4
 148|     for (px, py, pz, pw, ph, pd) in placed:
 149|         if abs(y - (py + ph)) > 1e-3:
 150|             continue
 151|         for j, (vx, vz) in enumerate(four):
 152|             if px <= vx <= px + pw and pz <= vz <= pz + pd:
 153|                 covered[j] = True
 154|     if not all(covered):
 155|         return False
 156|     return centre_ok

 159| def _is_accessible(x, y, z, w, h, d, seq, placed_full):
 160|     """Soft-LIFO accessibility — copy of ``Bin._isAccessibleAt`` logic.
 161| 
 162|     Returns True unless a higher-sequence box is *in front* (closer to the
 163|     door / higher z) in the same width lane and height shelf.
 164|     """
 165|     if seq is None:
 166|         return True
 167|     ix0 = x; ix1 = x + w
 168|     iy0 = y; iy1 = y + h
 169|     iz1 = z + d  # candidate's front face
 170|     for pb in placed_full:
 171|         oseq = pb["seq"]
 172|         if oseq is None or float(oseq) <= float(seq):
 173|             continue
 174|         jx0 = pb["x"]; jx1 = jx0 + pb["w"]
 175|         jy0 = pb["y"]; jy1 = jy0 + pb["h"]
 176|         jz0 = pb["z"]
 177|         if not (ix0 < jx1 - EPS and jx0 < ix1 - EPS):
 178|             continue
 179|         if not (iy0 < jy1 - EPS and jy0 < iy1 - EPS):
 180|             continue
 181|         if jz0 >= iz1 - 5e-4:
 182|             return False
 183|     return True

 190| @dataclass
 191| class _EPSet:
 192|     """Manages the set of extreme points (candidate placement corners)."""
 193|     width: float
 194|     height: float
 195|     depth: float
 196|     points: list = field(default_factory=list)  # (x, y, z) in cm
 197| 
 198|     def init(self):
 199|         """Seed with the container origin — the first placement corner."""
 200|         self.points = [(0.0, 0.0, 0.0)]
 201| 
 202|     def add_corners(self, x, y, z, w, h, d):
 203|         """Add the 8 corners of a placed box as candidate EPs."""
 204|         for cx in (x, x + w):
 205|             for cy in (y, y + h):
 206|                 for cz in (z, z + d):
 207|                     self.points.append((cx, cy, cz))
 208| 
 209|     def prune(self, placed_boxes):
 210|         """Remove EPs interior to a placed box or near-duplicated."""
 211|         filtered = []
 212|         for (px, py, pz) in self.points:
 213|             interior = False
 214|             for (bx, by, bz, bw, bh, bd) in placed_boxes:
 215|                 if (bx - EPS < px < bx + bw - EPS and
 216|                     by - EPS < py < by + bh - EPS and
 217|                     bz - EPS < pz < bz + bd - EPS):
 218|                     interior = True
 219|                     break
 220|             if not interior:
 221|                 filtered.append((px, py, pz))
 222|         unique = []
 223|         for p in filtered:
 224|             dup = False
 225|             for u in unique:
 226|                 if (abs(p[0] - u[0]) < EPS and
 227|                     abs(p[1] - u[1]) < EPS and
 228|                     abs(p[2] - u[2]) < EPS):
 229|                     dup = True
 230|                     break
 231|             if not dup:
 232|                 unique.append(p)
 233|         self.points = unique

 261| def _build_items_from_manifest(manifest):
 262|     """Expand the app.py manifest into a flat list of ``_PlacedItem``
 263|     templates (pre-placement; each copy is a distinct object)."""
 264|     templates = []
 265|     for m in manifest:
 266|         w = float(m["w"]) * 100   # m -> cm
 267|         h = float(m["h"]) * 100
 268|         d = float(m["d"]) * 100
 269|         qty = int(m.get("quantity", 1))
 270|         qty = qty if qty > 0 else 1
 271|         for i in range(qty):
 272|             templates.append(_PlacedItem(
 273|                 name=f'{m["name"]} #{i+1}',
 274|                 partno=f'ITEM-{m["name"]}#{i+1}',
 275|                 x=0.0, y=0.0, z=0.0,
 276|                 w=w, h=h, d=d,
 277|                 weight=float(m["weight"]),
 278|                 rotation_type=RotationType.RT_WHD,
 279|                 max_load=float(m["max_load"]) if math.isfinite(m["max_load"]) else float("inf"),
 280|                 seq=float(m["sequence"]),
 281|                 color=m.get("color", "#1f77b4"),
 282|             ))
 283|     return templates

 286| def _best_fit_place(item, ep_set, bin_w, bin_h, bin_d, support_ratio,
 287|                     max_weight, current_weight, placed_boxes, rotations,
 288|                     allow_lifo):
 289|     """Try every (EP, rotation); commit the best-fit (min residual).
 290| 
 291|     Returns True if placed, False otherwise.  Mutates ``placed_boxes``
 292|     and ``ep_set``.
 293|     """
 294|     best_residual = None
 295|     best_candidate = None
 296|     for (px, py, pz) in ep_set.points:
 297|         for rt in rotations:
 298|             w, h, d = _rot_dim(item.w, item.h, item.d, rt)
 299|             # Bounds check
 300|             if (px < -EPS or py < -EPS or pz < -EPS or
 301|                 px + w > bin_w + EPS or py + h > bin_h + EPS or
 302|                 pz + d > bin_d + EPS):
 303|                 continue
 304|             cand_aabb = (px, py, pz, w, h, d)
 305|             # Collision check
 306|             overlap = False
 307|             for pb in placed_boxes:
 308|                 if _aabb_overlap(cand_aabb, pb.aabb()):
 309|                     overlap = True
 310|                     break
 311|             if overlap:
 312|                 continue
 313|             # Stability check
 314|             floor_placed = [pb.aabb() for pb in placed_boxes]
 315|             if not _is_supported(px, py, pz, w, h, d, floor_placed, support_ratio):
 316|                 continue
 317|             # Weight check
 318|             if current_weight + item.weight > max_weight:
 319|                 continue
 320|             # Soft-LIFO accessibility
 321|             if allow_lifo:
 322|                 placed_full = [
 323|                     {"x": pb.x, "y": pb.y, "z": pb.z,
 324|                      "w": pb.w, "h": pb.h, "d": pb.d, "seq": pb.seq}
 325|                     for pb in placed_boxes
 326|                 ]
 327|                 if not _is_accessible(px, py, pz, w, h, d, item.seq,
 328|                                       placed_full):
 329|                     continue
 330|             # Best-fit merit: minimum residual across all three axes
 331|             rx = bin_w - (px + w)
 332|             ry = bin_h - (py + h)
 333|             rz = bin_d - (pz + d)
 334|             residual = min(rx, ry, rz)
 335|             if best_residual is None or residual < best_residual:
 336|                 best_residual = residual
 337|                 best_candidate = (px, py, pz, w, h, d, rt)
 338| 
 339|     if best_candidate is None:
 340|         return False
 341| 
 342|     px, py, pz, w, h, d, rt = best_candidate
 343|     item.x, item.y, item.z = px, py, pz
 344|     item.w, item.h, item.d = w, h, d
 345|     item.rotation_type = rt
 346|     placed_boxes.append(item)
 347|     ep_set.add_corners(px, py, pz, w, h, d)
 348|     return True

 351| def solve_sardine(manifest, truck_w, truck_h, truck_d, truck_weight,
 352|                   prioritize_sequence=False,
 353|                   support_surface_ratio=DEFAULT_SUPPORT_RATIO):
 354|     """Run the Extreme Point Insertion heuristic.
 355| 
 356|     Parameters mirror ``solver_classic.adapter.solve``.
 357|     """
 358|     truck = Truck(name="Truck", width=truck_w, height=truck_h,
 359|                   depth=truck_d, max_weight=truck_weight)
 360| 
 361|     # Work in cm internally.
 362|     bin_w = truck_w * 100
 363|     bin_h = truck_h * 100
 364|     bin_d = truck_d * 100
 365|     max_weight = truck_weight
 366| 
 367|     items = _build_items_from_manifest(manifest)
 368| 
 369|     # ── Ordering ──
 370|     if prioritize_sequence:
 371|         items.sort(key=lambda it: (-it.seq, -it.weight,
 372|                                     -(it.w * it.h * it.d)))
 373|     else:
 374|         items.sort(key=lambda it: (
 375|             -it.seq,
 376|             it.max_load if math.isfinite(it.max_load) else 0,
 377|             -(it.w * it.h * it.d),
 378|             -it.weight,
 379|         ))
 380| 
 381|     rotations = SPIN_ONLY  # updown=False for all items, matching app.py
 382| 
 383|     ep_set = _EPSet(width=bin_w, height=bin_h, depth=bin_d)
 384|     ep_set.init()
 385| 
 386|     placed = []
 387|     unfitted = []
 388|     weight_accum = 0.0
 389| 
 390|     for item in items:
 391|         placed_now = _best_fit_place(
 392|             item, ep_set,
 393|             bin_w, bin_h, bin_d,
 394|             support_surface_ratio,
 395|             max_weight, weight_accum,
 396|             placed, rotations,
 397|             allow_lifo=prioritize_sequence,
 398|         )
 399|         if placed_now:
 400|             weight_accum += item.weight
 401|             ep_set.prune([pb.aabb() for pb in placed])
 402|         else:
 403|             unfitted.append(item)
 404| 
 405|     # ── Build Placement list (cm -> m) ──
 406|     packed = [
 407|         Placement(
 408|             name=pb.name,
 409|             partno=pb.partno,
 410|             x=pb.x / 100.0,
 411|             y=pb.y / 100.0,
 412|             z=pb.z / 100.0,
 413|             w=pb.w / 100.0,
 414|             h=pb.h / 100.0,
 415|             d=pb.d / 100.0,
 416|             weight=pb.weight,
 417|             rotation_type=pb.rotation_type,
 418|             max_load=pb.max_load,
 419|             color=pb.color,
 420|         )
 421|         for pb in placed
 422|     ]
 423|     unfitted_placements = [
 424|         Placement(
 425|             name=u.name,
 426|             partno=u.partno,
 427|             x=0.0, y=0.0, z=0.0,
 428|             w=u.w / 100.0, h=u.h / 100.0, d=u.d / 100.0,
 429|             weight=u.weight,
 430|             rotation_type=u.rotation_type,
 431|             max_load=u.max_load,
 432|             color=u.color,
 433|         )
 434|         for u in unfitted
 435|     ]
 436| 
 437|     sol = Solution(
 438|         truck=truck,
 439|         packed=packed,
 440|         unfitted=unfitted_placements,
 441|         engine="sardine",
 442|         strategy="epi-baseline",
 443|         extra={
 444|             "method": "ExtremePointInsertion",
 445|             "prioritize_sequence": prioritize_sequence,
 446|             "ep_count": len(ep_set.points),
 447|         },
 448|     )
 449|     return sol
 450|

 452| def solve(manifest, truck_w, truck_h, truck_d, truck_weight,
 453|           prioritize_sequence=False,
 454|           support_surface_ratio=DEFAULT_SUPPORT_RATIO,
 455|           **kwargs):
 456|     """Thin wrapper matching the package __init__ export convention."""
 457|     return solve_sardine(
 458|         manifest, truck_w, truck_h, truck_d, truck_weight,
 459|         prioritize_sequence=prioritize_sequence,
 460|         support_surface_ratio=support_surface_ratio,
 461|     )
