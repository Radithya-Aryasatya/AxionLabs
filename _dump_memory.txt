   1| """
   2| solver_sardine.memory
   3| =====================
   4| Disk-backed persistence for the Sardine-Can AI solver so the AI's learning /
   5| history survives Streamlit restarts, refreshes, and new browser windows.
   6| 
   7| Layout (created lazily inside ``ai_progress/`` next to ``app.py``)::
   8| 
   9|     ai_progress/
  10|       config.json          # {improve_enabled, budget_seconds, ...}
  11|       history.jsonl        # one JSON line per packing run
  12|       best_solutions/      # {manifest_hash}.json  best known layout
  13|       learned_weights.json # {hash: {strategy: {weight, wins, losses}}}
  14|       README.md
  15|       .gitkeep
  16| 
  17| Every public function degrades gracefully on IO / permission errors - the
  18| solver is always allowed to fall back to its in-memory baseline.
  19| """

  48| def ensure_dirs():
  49|     """Create ``ai_progress/`` + ``best_solutions/`` and seed docs."""
  50|     try:
  51|         os.makedirs(BEST_DIR, exist_ok=True)
  52|     except OSError:
  53|         return
  54|     for name, content in (("README.md", _README), (".gitkeep", "")):
  55|         path = os.path.join(PROGRESS_DIR, name)
  56|         if not os.path.exists(path):
  57|             try:
  58|                 with open(path, "w", encoding="utf-8") as f:
  59|                     f.write(content)
  60|             except OSError:
  61|                 pass

  64| def _atomic_write(path, data):
  65|     """Write text to ``path`` atomically (temp file + os.replace)."""
  66|     tmp = path + ".tmp"
  67|     os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
  68|     try:
  69|         with open(tmp, "w", encoding="utf-8") as f:
  70|             f.write(data)
  71|             f.flush()
  72|             os.fsync(f.fileno())
  73|         os.replace(tmp, path)
  74|     except OSError as e:
  75|         logger.warning("atomic write failed for %s: %s", path, e)
  76| 
  77| 
  78| def _atomic_write_json(path, obj):
  79|     _atomic_write(path, json.dumps(obj, indent=2, default=str))
  80| 
  81| 
  82| def _read_json(path, default):
  83|     """Read a JSON file, returning ``default`` on any failure."""
  84|     try:
  85|         with open(path, encoding="utf-8") as f:
  86|             return json.load(f)
  87|     except (OSError, json.JSONDecodeError, ValueError):
  88|         return default

 103| def load_config() -> dict:
 104|     """Return persisted config merged over defaults. Never raises."""
 105|     cfg = _read_json(_CONFIG_PATH, {})
 106|     if not isinstance(cfg, dict):
 107|         cfg = {}
 108|     return {**_DEFAULT_CONFIG, **cfg}
 109| 
 110| 
 111| def save_config(cfg: dict):
 112|     """Persist user config atomically. No-raise on failure."""
 113|     ensure_dirs()
 114|     try:
 115|         with _lock:
 116|             _atomic_write_json(_CONFIG_PATH, {**_DEFAULT_CONFIG, **(cfg or {})})
 117|     except Exception as e:  # noqa: BLE001 - persistence must never break UI
 118|         logger.warning("save_config failed: %s", e)

 143| def record_strategy_result(sig: str, strategy: str, won: bool,
 144|                            score: float = 0.0):
 145|     """Update the win/loss record (and derived weight) for one strategy.
 146| 
 147|     ``weight`` starts at 0.5 and moves toward 1.0 on wins / 0.0 on losses
 148|     with an exponential moving average, so recent results dominate but a
 149|     single unlucky run cannot permanently blacklist a strategy.
 150|     """
 151|     ensure_dirs()
 152|     try:
 153|         with _lock:
 154|             weights = load_learned_weights()
 155|             bucket = weights.setdefault(sig, {})
 156|             entry = bucket.setdefault(
 157|                 strategy, {"weight": 0.5, "wins": 0, "losses": 0,
 158|                            "best_score": 0.0})

 167| def load_best_solution(sig: str):
 168|     """Load the best stored layout for a manifest hash, or None."""

 173| def save_best_solution(sig: str, solution_dict: dict, score: float) -> bool:
 174|     """Store a layout only if it beats the score already on disk.
 175| 
 176|     Returns True when the new layout was written.
 177|     """
 178|     ensure_dirs()
 179|     try:
 180|         with _lock:
 181|             prev = _read_json(_best_path(sig), None)
 182|             if isinstance(prev, dict) and float(prev.get("score", 0)) >= float(score):
 183|                 return False
 184|             payload = dict(solution_dict)
 185|             payload["score"] = float(score)
 186|             _atomic_write_json(_best_path(sig), payload)
 187|             return True
 188|     except Exception as e:  # noqa: BLE001
 189|         logger.warning("save_best_solution failed: %s", e)
 190|         return False

 227| def append_history(record: RunRecord):
 228|     """Append one line to ``history.jsonl`` (crash-tolerant)."""
 229|     ensure_dirs()
 230|     line = json.dumps(record.to_dict(), default=str) + "\n"
 231|     try:
 232|         with _lock:
 233|             with open(_history_path(), "a", encoding="utf-8") as f:
 234|                 f.write(line)
 235|                 f.flush()
 236|     except OSError as e:
 237|         logger.warning("append_history failed: %s", e)

 240| def load_history(limit: int | None = None) -> list:
 241|     """Load history records (skipping corrupt lines). Newest first."""
 242|     records = []
 243|     try:
 244|         with open(_history_path(), encoding="utf-8") as f:
 245|             for line in f:
 246|                 line = line.strip()
 247|                 if not line:
 248|                     continue
 249|                 try:
 250|                     records.append(json.loads(line))
 251|                 except json.JSONDecodeError:
 252|                     continue
 253|     except OSError:
 254|         return []
 255|     records.reverse()
 256|     return records[:limit] if limit else records

 281| def manifest_hash(manifest: list[dict], truck_dims: tuple,
 282|                   truck_weight: float) -> str:
 283|     """Stable hash of the full input so history/best are keyed reliably."""
 284|     payload = {
 285|         "truck": {
 286|             "w": float(truck_dims[0]), "h": float(truck_dims[1]),
 287|             "d": float(truck_dims[2]), "max_weight": float(truck_weight),
 288|         },
 289|         "items": sorted(
 290|             (
 291|                 {
 292|                     "name": m["name"],
 293|                     "w": round(float(m["w"]), 6),
 294|                     "h": round(float(m["h"]), 6),
 295|                     "d": round(float(m["d"]), 6),
 296|                     "weight": round(float(m["weight"]), 6),
 297|                     "quantity": int(m.get("quantity", 1)),
 298|                     "sequence": int(m.get("sequence", 0)),
 299|                 }
 300|                 for m in manifest
 301|                 if m.get("name") != "_truck_meta"
 302|             ),
 303|             key=lambda d: (d["sequence"], d["name"]),
 304|         ),
 305|     }
 306|     raw = json.dumps(payload, sort_keys=True, default=str)
 307|     return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:16]

 314| def hydrate_session_state(st_session):
 315|     """Bulk-load persisted data into ``st.session_state`` on startup.
 316| 
 317|     Called at the top of ``app.py`` so a browser refresh or a brand-new
 318|     window immediately sees the AI's previous progress.
 319|     """
 320|     try:
 321|         if "ai_config" not in st_session:
 322|             st_session.ai_config = load_config()
 323|         if "ai_history" not in st_session:
 324|             st_session.ai_history = load_history()
 325|         if "ai_weights" not in st_session:
 326|             st_session.ai_weights = load_learned_weights()
 327|         st_session.ai_loaded = True
 328|     except Exception as e:  # noqa: BLE001
 329|         logger.warning("hydrate_session_state failed: %s", e)
